\begin{frame}{Basic mechanism of property tax}

\begin{enumerate}
    \item \textbf{Assessment:} Local \textit{assessors} estimate \textbf{market value} of properties.
    \medskip

    \item \textbf{Base Adjustments:} Qualifying adjustments apply to assessed market value.
    \begin{itemize}
        \item E.g., Homestead exceptions, veterans, etc.
    \end{itemize}
    \medskip
    % Principal-residence, senior citizens, veterans

    \item \textbf{Tax Rate:} Local governments (city, county, etc.) set a tax rate.
    \begin{itemize}
        \item E.g., Cambridge: $\sim$ 0.55\%, Boston: $\sim$1.05\%, Chicago: $\sim$2\%.
    \end{itemize}
    \medskip

    \item \textbf{Tax Bill:} Property Tax = Adjusted Value $\times$ Tax Rate
\end{enumerate}

\begin{frame}{ML appears naturally}

\begin{itemize}
    \item Mass appraisal uses \textbf{many features and interactions}; housing markets are nonlinear.
    \medskip
    \item Assessor offices increasingly adopt \textbf{flexible ML learners} for the valuation step.
    \begin{itemize}
        \item Regularized linear models $\rightarrow$ \textbf{tree ensembles} (XGBoost, LightGBM) $\rightarrow$ neural networks.
        \item Gradient-boosted trees are now among the \underline{most common production learners}.
    \end{itemize}
\end{itemize}

\medskip

\begin{columns}[T,onlytextwidth]
\column{0.48\textwidth}
\begin{block}{\emoji{white-check-mark} What flexible ML buys}
\begin{itemize}
    \item Stronger \textbf{predictive accuracy} on tabular data.
    \item Captures \textbf{nonlinear market structure}.
    \item Scales to large parcel inventories.
\end{itemize}
\end{block}

\column{0.48\textwidth}
\begin{block}{\emoji{warning} What it does not solve}
\begin{itemize}
    \item Explainability \& defensibility.
    \item \textbf{Fairness/equity} in assessment.
    \item Operational feasibility.
\end{itemize}
\end{block}
\end{columns}

\begin{frame}{Fairness in assessment}

\begin{columns}[T,onlytextwidth]

\vspace{-1mm}
\[
r_i := \frac{\widehat{P}_i}{P_i} = \frac{\text{Assessed value}}{\text{Sale price}}.

\section{Proposed method}


% ----------------------------------------------------------
% Slide 11 -- Main idea: fairness as regularization
% ----------------------------------------------------------
\begin{frame}{Main idea: penalize value-dependent error patterns}

\begin{columns}[T,onlytextwidth]
\column{0.48\textwidth}
\begin{block}{Old paradigm}
Minimize in-sample prediction error:
\vspace{-1mm}
\[
f_{\text{base}} \in \arg\min_{f\in\H}\; L(f; X, y).
\]
\begin{itemize}
    \item Pure accuracy objective.
    \item \emoji{warning} Ignores value-dependent ratio drift.
\end{itemize}
\end{block}

\column{0.48\textwidth}
\begin{block}{Regressivity-aware paradigm}
Add a \textbf{dependence penalty} $\Psi$:
\vspace{-1mm}
\[
f_{\text{dep}} \in \arg\min_{f\in\H}\big\{ L(f; X, y) + \rho\,\Psi(r, y) \big\}.
\]
\begin{itemize}
    \item One extra hyperparameter $\rho \ge 0$.
    \item Recovers the baseline at $\rho=0$.
\end{itemize}
\end{block}
\end{columns}

\vspace{3mm}

\begin{block}{\emoji{light-bulb} Three moves}
\begin{enumerate}
    \item Start from a \textbf{strong predictive learner} (e.g., LightGBM).
    \item Add a penalty $\Psi$ that \underline{discourages value-dependent error patterns}.
    \item Tune $\rho$ to navigate the \textbf{accuracy--equity tradeoff}.
\end{enumerate}
\end{block}

\end{frame}


% ----------------------------------------------------------
% Slide 12 -- Why covariance, and why log-space
% ----------------------------------------------------------
\begin{frame}{Why covariance, and why log-space}

\begin{block}{\emoji{pushpin} A natural statistical target}
Train on the log-price scale: $y=\log P$, $\hat y = \log \widehat P$. The log-scale residual is a log-ratio:
\vspace{-1mm}
\[
e_i \;=\; \hat y_i - y_i \;=\; \log\!\left(\frac{\widehat P_i}{P_i}\right) \;=\; \log r_i.
\]
\end{block}

\vspace{-1mm}

\begin{itemize}
    \item \textbf{Dependence} between $e$ and $y$ $\;\Leftrightarrow\;$ \underline{value-dependent ratio trend}.
    \medskip
    \item \textbf{Covariance} $\Cov(e, y)$ captures the \textbf{first-order (linear) trend} visible in the ratio--price plot.
    \medskip
    \item Flattening $\Cov(e, y) \to 0$ flattens the main \underline{price-tier trend} behind PRD, PRB, VEI.
\end{itemize}

\vspace{2mm}

\begin{block}{\emoji{light-bulb} Intuition in one line}
\begin{center}
``\textit{If residuals drift with price, then ratios drift with price.\\ Penalize one covariance, flatten the whole trend.}''
\end{center}
\end{block}

\end{frame}


% ----------------------------------------------------------
% Slide 13 -- Two penalty variants
% ----------------------------------------------------------
\begin{frame}{Two penalty variants}

\vspace{-1mm}
\begin{columns}[T,onlytextwidth]
\column{0.49\textwidth}
\begin{block}{\textbf{Direct}: squared covariance}
\vspace{-4mm}
\[
\Psi_{\text{cov}} = \widehat{\Cov}(e, y)^2
= \Big(\tfrac{1}{n}\sum_i (e_i - \bar e)(y_i - \bar y)\Big)^2
\]
\begin{itemize}
    \item \emoji{white-check-mark} Closest to the \textbf{conceptual target}.
    \item \emoji{warning} \textbf{Non-separable} across samples (global coupling).
    \item \emoji{warning} Dense Hessian $\Rightarrow$ diagonal approximation or custom leaf solver.
\end{itemize}
\end{block}

\column{0.49\textwidth}
\begin{block}{\textbf{Surrogate}: separable upper bound}
\vspace{-4mm}
\[
\Psi_{\text{surr}} = \tfrac{1}{n}\sum_{i=1}^n (e_i)^2\,(y_i - \bar y)^2
\]
\begin{itemize}
    \item \emoji{white-check-mark} Separable \& \textbf{sample-additive}.
    \item \emoji{white-check-mark} \textbf{Plug-and-play} into LightGBM / XGBoost custom-objective APIs.
    \item \emoji{white-check-mark} Weights residual size by distance from $\bar y$.
\end{itemize}
\end{block}
\end{columns}

\vspace{3mm}

\begin{center}
\small
\begin{tabular}{ll}
\textbf{Direct} $\Psi_{\text{cov}}$   & $\rightarrow$ \emph{conceptual fidelity}.\\
\textbf{Surrogate} $\Psi_{\text{surr}}$ & $\rightarrow$ \emph{implementation convenience} (our default).
\end{tabular}
\end{center}

{\footnotesize\itshape
Both are governed by a single $\rho \ge 0$; we refer to both as ``covariance'' penalties. Derivations, Hessians, and boosting-API details are in the appendix.
}

\end{frame}

\begin{frame}{Deployment logic: constrained calibration}

\begin{block}{\emoji{bullseye} Multi-objective selection, assessor-style}
In deployment, \textbf{multiple diagnostics matter simultaneously} (accuracy, COD, PRD, PRB, VEI, \dots).\\
Calibration is naturally a \underline{constrained} selection problem: pick the strongest predictive candidate \emph{among those satisfying ratio-study tolerances}.
\end{block}

\vspace{-1mm}

\renewcommand{\C}{\mathcal{C}}
\renewcommand{\V}{\mathcal{V}}

\begin{columns}[T,onlytextwidth]
\column{0.56\textwidth}

\textbf{Setup.}
\begin{itemize}
    \item $c \in \C$: candidate configurations (model family, hyperparameters, penalty $\rho$).
    \item $v \in \V$: rolling-origin validation folds.
    \item For each $(c,v)$: prediction error $\mathcal{E}_v(c)$ and an equity metric $\mathcal{F}_v(c)$.
\end{itemize}

\medskip

\textbf{Selection rule:}
\vspace{-1mm}
\[
c^* \in \argmin_{c \in \C}\; \frac{1}{|\V|}\sum_{v\in\V} \mathcal{E}_v(c)
\quad
\text{s.t.}\; \mathcal{F}_v(c) \le \tau,\;\; \forall v\in\V.
\]

\column{0.42\textwidth}
\begin{block}{\emoji{white-check-mark} Properties}
\begin{itemize}
    \item Add \textbf{any number} of equity constraints (PRD, PRB, VEI, COD, \dots).
    \item $c^*$ is a \textbf{Pareto-optimal} candidate under the chosen tolerances.
    \item Same logic as the office's \underline{ratio-study compliance check}.
\end{itemize}
\end{block}
\end{columns}

\end{frame}

\begin{frame}{Limitations and honest scope}

\begin{block}{What the method does \emph{not} (yet) do}
\begin{enumerate}
    \medskip
    \item Does \textbf{not optimize PRD / PRB / VEI directly}; it controls a covariance that is directionally aligned with them.
    \medskip
    \item Provides \textbf{no out-of-sample guarantees} on ratio-study compliance; this is why we use constrained calibration across folds.
    \medskip
    \item Covariance is a \textbf{first-order summary} (linear dependence); nonmonotone local distortions can cancel.
    \medskip
    \item Penalties are \textbf{global}; not \underline{submarket- or geography-aware}.
    \medskip
    \item The CCAO pipeline we use is a \textbf{translation} of the production code; absolute levels should be read in that context.
    \medskip
\end{enumerate}

\begin{frame}{Appendix A1: Metric definitions (accuracy, horizontal \& vertical equity)}

Assessment-ratio statistics:
\footnotesize
\setlength{\abovedisplayskip}{3pt}
\setlength{\belowdisplayskip}{3pt}
\setlength{\abovedisplayshortskip}{2pt}
\setlength{\belowdisplayshortskip}{2pt}

\[
r_i := \frac{AV_i}{MV_i} = \frac{\widehat{MV}_i}{MV_i},\qquad
\tilde r := \mathrm{median}(r),\qquad
\bar r := \frac{1}{n}\sum_i r_i,\qquad
\bar r_w = \frac{\sum_i MV_i\,r_i}{\sum_i MV_i}
\]

\begin{columns}[T,onlytextwidth]
\column{0.48\textwidth}

\begin{block}{Accuracy}
\[
R^2 = 1 - \frac{\sum_i (MV_i - \widehat{MV}_i)^2}{\sum_i (MV_i - \overline{MV})^2},
\qquad
\mathrm{MAE} = \tfrac{1}{n}\sum_i |MV_i - \widehat{MV}_i|
\]
\end{block}

\begin{block}{Horizontal equity / dispersion (IAAO)}
\[
\mathrm{COD} = 100\cdot \frac{\tfrac{1}{n}\sum_i |r_i - \tilde r|}{\tilde r},
\qquad
\mathrm{COV} = \frac{\mathrm{SD}(r)}{\bar r}
\]
\end{block}

\column{0.49\textwidth}

\begin{block}{Vertical equity (IAAO \& CCAO)}
\[
\mathrm{VEI} \approx 100\cdot \frac{\mathrm{median}(r)_{\text{last dec.}} - \mathrm{median}(r)_{\text{first dec.}}}{\tilde r}
\]
{\scriptsize (VEI${<}0$ regressive; VEI${>}0$ progressive.)}
\medskip
\[
\mathrm{PRD} = \frac{\bar r}{\bar r_w},\qquad
\mathrm{PRB} = \mathrm{slope}\!\left(\tfrac{r_i - \tilde r}{\tilde r} \sim \log_2\!\Big(\tfrac{1}{2}(MV_i + AV_i/\tilde r)\Big)\right)
\]
\[
\mathrm{MKI} = \frac{\mathrm{Gini}(AV)}{\mathrm{Gini}(MV)}
\]
\end{block}
\end{columns}

\end{frame}

% A3 -- Implementation details
% ----------------------------------------------------------
\begin{frame}{Appendix A3: General-learner with covariance penalty}

\vspace{-1mm}
\begin{block}{Base learning problem}
For a hypothesis class $\H$ (linear, GBM, NN, \dots), standard learners solve
\vspace{-1mm}
\[
f^*(X) = \arg\min_{f\in\H}\; \mathcal{L}(f; X, y) = \arg\min_{f\in\H}\; J(f; X, y) + \Omega(f).
\]
\end{block}

\begin{block}{Covariance-penalized objective}
With ratio proxy $g(\hat y, y)$ and value transform $\phi(y)$:
\vspace{-1mm}
\[
\min_{f\in\H}\; \mathcal{L}(f; X, y) + \rho\, \big(\widehat{\Cov}(g(\hat y, y), \phi(y))\big)^2,
\]
or the constrained form
\vspace{-1mm}
\[
\min_{f\in\H}\; \mathcal{L}(f; X, y)
\quad\text{s.t.}\quad
\big(\widehat{\Cov}(g(\hat y, y), \phi(y))\big)^2 \le \varepsilon.
\]
\end{block}

\begin{columns}[T,onlytextwidth]
\column{0.48\textwidth}
\begin{block}{Choice of $g$ and $\phi$}
\footnotesize
Training on log-price $y=\log P$, $\hat y=f(X)$:
\begin{itemize}
    \item $g(\hat y, y) = \hat y - y$ (log-ratio; target 0), or
    \item $r = \hat y / y$ with $g = r$ (target 1).
    \item Start with $\phi(y) = y$ (log market value).
\end{itemize}
\end{block}

\column{0.48\textwidth}
\begin{block}{Coupling and separable surrogate}
\footnotesize
$\widehat{\Cov}(g,\phi)$ is \textbf{non-separable} (dense Hessian).
\vspace{-1mm}
\[
\widehat{\Cov}(g,\phi)^2 \le \tfrac{1}{n}\sum_{i=1}^n (g_i - \bar g)^2 (\phi_i - \bar\phi)^2
\]
$\Rightarrow$ \textbf{separable} upper bound; plugs into LightGBM / XGBoost custom-objective APIs via per-sample gradient and Hessian.
\end{block}
\end{columns}

\end{frame}

\begin{frame}{Appendix A4: Alternative selection criteria (Utopia / Nash)}

\begin{itemize}
    \item The constrained rule is our \textbf{recommended default}, but it has limits:
    \begin{itemize}
        \item Does not guarantee out-of-sample compliance.
        \item May be infeasible on some validation folds.
        \item May sacrifice too much accuracy when tolerances are strict.
    \end{itemize}
    \medskip
    \item \textbf{Alternative Pareto criteria} (used when tolerances are unknown / jointly infeasible):
    \begin{itemize}
        \item \textbf{Closest-to-Utopia:} minimize normalized distance to an ideal point that is best on every metric.
        \item \textbf{Nash / maximum-volume:} maximize the volume of a hyper-rectangle inscribed in the Pareto frontier.
    \end{itemize}
\end{itemize}

% ----------------------------------------------------------
% A5 -- Constrained-ML challenges and extensions
% ----------------------------------------------------------
\begin{frame}{Appendix A5: Why not constrain the learner directly?}

\begin{block}{\emoji{warning} Three challenges of fully constrained ML for assessment}
\begin{enumerate}
    \item \textbf{Feasibility / learnability (train):} hard constraints like ``PRB${=}0$'' may be infeasible or unstable in-sample, especially for simpler learners.
    \medskip
    \item \textbf{Generalization / robustness (test):} constraints tied to complex dependencies can fail out-of-sample (selection bias + small labeled sales population).
    \medskip
    \item \textbf{Implementation:} non-convex / non-differentiable ratio-study metrics (e.g., VEI) require substantially more sophisticated machinery.
\end{enumerate}
\end{block}

\medskip

\begin{block}{\emoji{light-bulb} Design choice}
Start with \textbf{simple dependence control} (linear / rank-1) that is robust with small samples; extend to \underline{nonlinear dependence} (HSIC, dCov, maximal correlation), \underline{submarket-aware} penalties, and \underline{interpretable learners} (EBM/GAM) in future work.
\end{block}

\end{frame}
