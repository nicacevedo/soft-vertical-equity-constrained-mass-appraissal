\newcommand{\frameZ}{Motivation: Regressivity in Mass Appraisal}
\begin{frame}{\frameZ}
% \textbf{Our understanding:}
    \begin{enumerate}
        \item \textbf{Mass Appraisal} is a highly relevant \& sensitive topic. 
        
    \begin{itemize}
        \item Main source of revenue for local governments in the US.
        \item Affect almost everyone (sells, rent, public services, etc.)
    \end{itemize}
    \medskip
    \item \textbf{Fairness/equity} in this process a top priority \& has been well-studied (even before ML). Assessments should be \textit{fair} across properties of:
    \begin{itemize}
        \item Similar characteristics (\textit{Horizontal Equity)}. 
        \item Different price levels (\underline{\textit{Vertical Equity}}). 
    \end{itemize}
    \medskip
    \item \textbf{Regressivity} occurs when properties at the lower-end of the market tend to be over-assessed (assessment > market price) \&  the ones at the higher-end tend to be under-assessed (assessment < market price).
    \begin{itemize}
        \item Very common in real-world assessments.
        \item \textit{Progressivity} is the opposite case, but seems to be less common.
    \end{itemize}
    \end{enumerate}

\end{frame}


% \newcommand{\frameZ}{Motivation: Regressivity in Mass Appraisal}
\begin{frame}{\frameZ}
\textbf{Our contributions:}
\medskip
    \begin{enumerate}
        \item Add \textbf{minimal modifications} to general regression models used in practice (e.g., linear regression, boosted trees, etc.) for reducing regressivity, without 
        \begin{itemize}
            \item adding extra load to the workflow,
            \item changing the models used,
             \item affecting the interpretability of the results.
        \end{itemize}

        
    % \begin{itemize}
    %     \item Main source of revenue for local governments in the US.
    %     \item Affect almost everyone (sells, rent, public services, etc.)
    % \end{itemize}
    \medskip
    \item Add only \textbf{1 tunable extra hyperparameter}, that can be selected in a simple extra step (after model selection). 
    \medskip
    \item Propose an \textbf{additional (but optional) model selection} step, to secure \underline{better tradeoffs} between accuracy/dispersion \& regressivity constraints.
    \end{enumerate}
\end{frame}

   \emoji{bullseye} We aim for a \underline{middle-ground solution} that is
   \medskip
   \begin{itemize}
  \item Competitive in \textbf{accuracy} and \textbf{horizontal equity/dispersion}.
  \item Competitive in \textbf{vertical equity} .
  \item Ideally, improving metrics at least up to IAAO's \textbf{compliance ranges}.\\
   \end{itemize}

   \medskip

   \begin{block}{\emoji{light-bulb} \textbf{Solution Approach:} Covariance-Penalized Model}
       Take the \underline{best performing model} (accuracy \& dispersion), and \underline{add regressivity awareness}:
       
       % \medskip
       
       % Specifically,

        \medskip

\begin{enumerate}
    \item Take $\hat{y}=f(X)$, an accurate model (low error \& dispersion).
    \item Add a covariancce penalization term: $$ \text{Predictive Error} + \rho \cdot  \text{Cov(ratios, price)}^2$$
    \item Train the new model, minimizing the penalized objective with $\rho>0$.
    \item Select $\rho$ that achieves a middle-ground solution in validation sets.
\end{enumerate}
    % \item Re-calibrate to get the best model for the new objective.
% \end{enumerate}

   \end{block}

\begin{frame}{\frameJ}
\subsection{\frameJ}
\begin{block}{\emoji{warning} Tackling a \textbf{multi-objective problem}}
We aim to control \underline{3 objectives}:
\begin{center}    
Accuracy \(\leftrightarrow\) dispersion (horizontal equity) \(\leftrightarrow\) regressivity (vertical equity). 
\end{center}
Hard to characterize the true Pareto Frontier and tune in out-of-sample sets with limited labeled sales.
\end{block}

\begin{block}{\emoji{warning} Penalized/Constrained ML: two main challenges}
\begin{enumerate}
  \item \textbf{Feasibility / learnability (Train):} hard constraints like ``PRB \(=0\)'' (perfectly flat trend) \underline{can be infeasible or unstable} in-sample (especially for simpler models).
  \item \textbf{Generalization / robustness (Test):} constraints tied to complex dependencies can fail in out-of-sample sets (selection bias + small labeled fraction).
\end{enumerate}
\end{block}

\begin{block}{\emoji{light-bulb} Design principle / First approach}
Start with \textbf{simple dependence control} (linear / rank-1) that is robust with small samples, then extend.
\end{block}
\end{frame}

\vspace{-2mm}
\begin{block}{\emoji{pushpin} What we propose}
A \textbf{plug-in vertical-equity layer} for any learner:
\[
\text{loss} \;+\; \rho\cdot \widehat{\Cov}\big(g(\yh,y),y\big)^2,
\]
to reduce linear price-tier bias in assessment ratios. $g(\hat{y},y)$ can be either $\hat{y}/y$ (div) or $\hat{y}-y$ (diff). 
\end{block}

\begin{block}{\emoji{pushpin} Why it fits assessor workflows}
\begin{itemize}
  \item Works with existing model stacks (e.g. LightGBM).
  \item One knob \(\rho>0\) \& familiar diagnostics (ratio plot, PRD/PRB, COD).
  \item Provides a clear, auditable tradeoff curve for policy/operational choice.
\end{itemize}
\end{block}

\begin{block}{\emoji{pushpin} \textbf{Main takeaway}}
High accuracy does not guarantee vertical equity; we can enforce a practical notion of fairness-by-price-tier during training.
\end{block}
\end{frame}

\begin{frame}{\color{white} Appendix: General-learner with covariance constraint / penalty}
\vspace{-2mm}
\begin{block}{Base learning problem (the model already being used)}
Given hypothesis class \(\mathcal{H}\) (Linear Regression, Gradient Boosting, Neural Networks, etc.), regression models solve
\[
f^*(X)=\arg\min_{f\in\mathcal{H}} \; \mathcal{L}(f;X,y)
\;=\;
\arg\min_{f\in\mathcal{H}} \; J(f;X,y) + \Omega(f),
\]
with \(J\) empirical risk (typically MSE), \(\Omega\) complexity control (regularizer), and $f^*(X) = \hat{y}$ the estimator.
\end{block}

\begin{block}{Add a vertical-equity dependence control}
Let \(g(\yh,y)\) be some assessment ratio proxy and \(\phii(y)\) a value transform of the market value. We solve
\vspace{-2mm}
\[
\min_{f\in\mathcal{H}} \; \mathcal{L}(f;X,y)
\quad\text{s.t.}\quad
\big(\widehat{\Cov}(g(\yh,y),\phii(y))\big)^2 \le \varepsilon,
\]
or the soft penalty version:
\vspace{-2mm}
\[
\min_{f\in\mathcal{H}} \; \mathcal{L}(f;X,y)
\;+\;
\rho\big(\widehat{\Cov}(g(\yh,y),\phii(y))\big)^2.
\]
\footnotesize \textbf{Practical meaning:} take an existing assessor model, add one tunable term that discourages price-tier bias (linear trend).
\end{block}
\end{frame}

% 7. Note on coupling / separable surrogate (your slide)
% ==========================================================
\begin{frame}{\color{white} Appendix: choosing \(g,\phi\) and dealing with coupling}
\begin{columns}[T,onlytextwidth]
\column{0.5\textwidth}
\begin{block}{Choosing \(g(\yh,y)\) and \(\phii(y)\)}
Common training is in log space: \(y=\log(\MV)\), \(\yh=f(X)\).
\begin{itemize}
  % \item \textbf{Do not use} \(g(\yh,y)=\yh/y\) when \(y\) is a log (not a meaningful ratio).
  \item Log-space proxies:
  \[
  g(\yh,y)=\yh-y \quad \text{(log-ratio, target 0)},
  \]
  or a log-ratio proxy:
  \[
  r=\yh / y,\quad g(\yh,y)=r \quad \text{(target 1)}.
  \]
  \item We start with \(\phii(y)=y\) (log market value); extend in future work.
\end{itemize}
\end{block}

\column{0.47\textwidth}
\begin{block}{Coupling of the covariance term}
\[
\widehat{\Cov}(g,\phii)=\frac{1}{n}\sum_{i=1}^n (g_i-\bar g)(\phii_i-\bar\phii)
\]
\begin{itemize}
  \item \(\widehat{\Cov}(g,\phii)\) is \textbf{global} \(\Rightarrow\) non-separable.
  \item Exact Hessian is dense; many libraries expect per-sample Hessians.
\end{itemize}
\end{block}

\begin{block}{Separable surrogate (upper bound)}
Using a triangle-inequality/Jensen-type bound:
\[
\widehat{\Cov}(g,\phii)^2 \;\le\; \frac{1}{n}\sum_{i=1}^n (g_i-\bar g)^2(\phii_i-\bar\phii)^2.
\]
\end{block}
\end{columns}
\end{frame}
